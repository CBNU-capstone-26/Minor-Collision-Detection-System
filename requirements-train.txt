# ================================================================
# requirements-train.txt
# 물피도주 감지 "모델 학습 · 추론" 전용 Python 환경 (model/ = x3d-m)
# 재현 검증 환경: Python 3.11.15 (로컬 CPU) / Python 3.10 (학습 서버 GPU, RTX 3080)
#
# 이 파일은 ML 코드(train / predict / eval)만 실행하기 위한 최소 의존성입니다.
# 웹 서비스(FastAPI·Celery·MySQL·Redis)는 포함하지 않습니다 → 그건 루트 requirements.txt 사용.
# ================================================================
#
# -- 무엇을 위한 파일인가 -----------------------------------------
#   서비스·학습 모델은 model/(x3d-m)이고, 그 폴더 안에서 실행한다.
#   (예전 s3d 백본은 "s3d model/" 로 옮겨 git 에서 제외했다. slowfast model/ 은 사용 안 함)
#     (cd model && PRETRAINED=1 python3 -u main.py --mode train)    # 학습
#     (cd model && EVAL_WEIGHTS=../weights/<파일> python3 -u main.py --mode eval)
#     (cd model && python3 -u main.py --mode predict)                      # 단일 영상 CAM
#     (cd model && python3 ../check_probs.py ../weights/<파일>)            # 검증셋 확률 분포
#     (cd model && python3 trainplot.py ../outputs/trainlogs/*.csv)        # 학습 곡선 그래프
#   ※ 파이프(| tee)로 로그를 받을 땐 -u 를 붙여야 출력이 바로 보인다(버퍼링).
#
# -- 학습 서버 준비: tmux (시스템 패키지 — pip 이 아님) --------------
#   학습은 몇 시간 걸리는데, SSH 가 끊기면 그 셸에서 돌던 학습도 같이 죽는다.
#   tmux 세션 안에서 돌리면 접속이 끊겨도 학습은 계속되고 나중에 다시 붙을 수 있다.
#     sudo apt install tmux
#     tmux new -s train          # 세션을 만들고 그 안에서 학습 명령 실행
#     Ctrl+b 를 누른 뒤 d        # 세션에서 빠져나오기 (학습은 계속 돈다)
#     tmux attach -t train       # 다시 붙어서 진행 확인
#     tmux ls                    # 세션 목록
#
# -- torch / torchvision 은 아래 별도 스텝으로 설치 ----------------
#   PyTorch 전용 인덱스에서 받아야 하므로 이 파일에 넣지 않습니다.
#   (PyPI와 인덱스가 달라 한 파일에 섞으면 uv 인덱스 정책과 충돌 가능)
#
#   ※ 버전을 고정하지 말고 각 CUDA 인덱스가 제공하는 최신 쌍을 설치하는 것을 권장.
#     PyTorch는 오래된 CUDA 빌드를 순차 종료하므로, 특정 torch 버전이 특정 인덱스에
#     없을 수 있다. (예: cu121 인덱스는 torch 2.5.1 까지만 제공 → 2.12.0 설치 불가)
#     이 프로젝트 코드는 torch 2.5.1 이상이면 정상 동작한다(S3D·AMP·channels_last).
#
#   • 학습(GPU/CUDA): 서버 드라이버가 지원하는 CUDA에 맞는 인덱스 선택
#       (nvidia-smi 의 'CUDA Version' 확인 → cu121 / cu124 / cu126 / cu128 …)
#       pip install --index-url https://download.pytorch.org/whl/cu121 torch torchvision
#     → torch·torchvision 짝은 pip이 자동 해결 (cu121이면 2.5.1 + 0.20.1)
#     → config.py 의 TRAIN_DEVICE_TYPE="cuda"
#
#   • 추론(CPU, 서비스/로컬 검증):
#       pip install --index-url https://download.pytorch.org/whl/cpu torch torchvision
#     → config.py 의 INFER_DEVICE_TYPE 은 "cuda" 로 되어 있지만, GPU가 없으면
#       경고를 찍고 CPU로 자동 전환되므로 그대로 둬도 된다.
#     (로컬 검증 시 실제 설치본: torch 2.12.0+cpu / torchvision 0.27.0+cpu)
#
# -- 설치 순서 (로컬, CPU 추론 기준) -------------------------------
#   uv venv --python 3.11 venv311
#   uv pip install --python venv311 -r requirements-train.txt
#   uv pip install --python venv311 \
#       --index-url https://download.pytorch.org/whl/cpu torch==2.12.0 torchvision==0.27.0
#
#   (일반 venv/pip)
#   python3.11 -m venv venv311 && source venv311/bin/activate
#   pip install -r requirements-train.txt
#   pip install --index-url https://download.pytorch.org/whl/cpu torch torchvision
# ================================================================

# ---- 딥러닝 프레임워크 (별도 인덱스 스텝으로 설치; 위 헤더 참고) ----

# ---- 영상 처리 / 수치 연산 (dataset·predict_cam·evaluate) ----
# Python 버전에 따라 제공되는 최신 버전이 다르므로 하한(>=)만 지정해 이식성 확보.
# (정확 핀(==)은 특정 Python에서만 휠이 있어 resolution 실패 위험 — 예: numpy 2.4.x는 Python 3.11+ 전용)
# 로컬 검증(Python 3.11.15) 당시 실제 버전: numpy 2.4.6 / opencv 4.13.0.92 / pillow 12.2.0 / tqdm 4.68.1
numpy>=1.24                         # 수치 연산 (Python 3.10 → 2.2.x, 3.11+ → 2.4.x 등 자동 선택)
opencv-python-headless>=4.8         # cv2: 프레임 디코딩·크롭·CAM 오버레이·클립 렌더링
imageio-ffmpeg>=0.6.0               # 시스템 ffmpeg가 없을 때 쓰는 내장 ffmpeg (predict 모드 클립 H.264 변환)
pillow>=10.0                        # PIL: 데이터 증강(색상 지터 등) 시 이미지 변환
tqdm>=4.65                          # 평가 진행률 표시
